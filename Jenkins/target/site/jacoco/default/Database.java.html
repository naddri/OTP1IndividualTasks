import java.sql.*;
import java.util.ArrayList;
import java.util.List;

public class Database {

    public record Trip(int id, String name) {
        @Override
        public String toString() {
            return name;
        }
    }

    private final Connection conn;

    public Database(String url) throws SQLException {
        conn = DriverManager.getConnection(url);
        try (Statement st = conn.createStatement()) {
            st.execute("CREATE TABLE IF NOT EXISTS trip (id INTEGER PRIMARY KEY AUTOINCREMENT, name TEXT NOT NULL)");
            st.execute("CREATE TABLE IF NOT EXISTS measurement (id INTEGER PRIMARY KEY AUTOINCREMENT, "
                    + "trip_id INTEGER NOT NULL, distance REAL, time REAL, speed REAL, "
                    + "FOREIGN KEY (trip_id) REFERENCES trip(id))");
        }
    }

    public int addTrip(String name) throws SQLException {
        try (PreparedStatement ps = conn.prepareStatement("INSERT INTO trip(name) VALUES (?)",
                Statement.RETURN_GENERATED_KEYS)) {
            ps.setString(1, name);
            ps.executeUpdate();
            ResultSet rs = ps.getGeneratedKeys();
            rs.next();
            return rs.getInt(1);
        }
    }

    public List<Trip> getTrips() throws SQLException {
        List<Trip> list = new ArrayList<>();
        try (Statement st = conn.createStatement();
             ResultSet rs = st.executeQuery("SELECT id, name FROM trip ORDER BY id")) {
            while (rs.next()) {
                list.add(new Trip(rs.getInt("id"), rs.getString("name")));
            }
        }
        return list;
    }

    // speed = distance / time (time in hours)
    public double addMeasurement(int tripId, double distance, double time) throws SQLException {
        if (time <= 0) {
            throw new IllegalArgumentException("time must be positive");
        }
        double speed = distance / time;
        try (PreparedStatement ps = conn.prepareStatement(
                "INSERT INTO measurement(trip_id, distance, time, speed) VALUES (?,?,?,?)")) {
            ps.setInt(1, tripId);
            ps.setDouble(2, distance);
            ps.setDouble(3, time);
            ps.setDouble(4, speed);
            ps.executeUpdate();
        }
        return speed;
    }

    public List<String> getMeasurements(int tripId) throws SQLException {
        List<String> list = new ArrayList<>();
        try (PreparedStatement ps = conn.prepareStatement(
                "SELECT distance, time, speed FROM measurement WHERE trip_id = ? ORDER BY id")) {
            ps.setInt(1, tripId);
            ResultSet rs = ps.executeQuery();
            while (rs.next()) {
                list.add(String.format("%.2f km / %.2f h = %.2f km/h",
                        rs.getDouble("distance"), rs.getDouble("time"), rs.getDouble("speed")));
            }
        }
        return list;
    }
}
